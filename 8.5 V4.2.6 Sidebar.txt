  <script>
    // --- collapsibles ---
    document.querySelectorAll('.collapsible').forEach(btn=>{
      btn.addEventListener('click',()=>{
        btn.classList.toggle('active');
        const c = btn.nextElementSibling;
        if(c.style.maxHeight){ c.style.maxHeight = null; }
        else { c.style.maxHeight = c.scrollHeight + 'px'; }
      });
    });

    // --- init from backend ---
    google.script.run.withSuccessHandler(initSidebar).getSidebarData();

    function initSidebar(data){
      // Courses
      const courseSel = document.getElementById('course'); courseSel.innerHTML='';
      (data.courses||[]).forEach(c=>{
        const o=document.createElement('option'); o.value=c; o.textContent=c; courseSel.appendChild(o);
      });
      // Checkboxes
      fillCheckGrid('difficulties', data.difficulties||[]);
      fillCheckGrid('types', data.types||[]);
      fillCheckGrid('correct-status', data.correctStatuses||[]);
      // Quotas
      buildQuotaInputs('type-quotas', data.types||[]);
      buildQuotaInputs('diff-quotas', data.difficulties||[]);
      // Open Quotas by default
      openSection('btnQuotas','sectQuotas');
      // Listeners for highlight
      attachActivityListeners();
      updateActiveHighlights();
      // Wire buttons
      wireButtons();
    }

    // --- UI builders ---
    function fillCheckGrid(containerId, items){
      const div=document.getElementById(containerId); div.innerHTML='';
      items.forEach(item=>{
        const lbl=document.createElement('label');
        const cb=document.createElement('input');
        cb.type='checkbox'; cb.value=item;
        lbl.appendChild(cb); lbl.appendChild(document.createTextNode(' '+item));
        div.appendChild(lbl);
      });
    }
    function buildQuotaInputs(containerId, items){
      const div=document.getElementById(containerId); div.innerHTML='';
      items.forEach(item=>{
        const id = keyify(item)+'-quota';
        const g=document.createElement('div'); g.className='form-group';
        g.innerHTML=`<label>${item}</label><input type="number" id="${id}" min="0">`;
        div.appendChild(g);
      });
    }
    function openSection(btnId, sectId){
      const btn=document.getElementById(btnId), sect=document.getElementById(sectId);
      if(btn && sect){ btn.classList.add('active'); sect.style.maxHeight = sect.scrollHeight + 'px'; }
    }

    // --- highlight logic ---
    function attachActivityListeners(){
      document.querySelectorAll('input,select,textarea').forEach(el=>{
        el.addEventListener('change', updateActiveHighlights);
        el.addEventListener('input', updateActiveHighlights);
      });
    }
    function updateActiveHighlights(){
      document.querySelectorAll('.collapsible').forEach(btn=>{
        const sect = btn.nextElementSibling;
        const inputs = sect.querySelectorAll('input,select,textarea');
        const active = Array.from(inputs).some(el=>{
          if(el.type==='checkbox'||el.type==='radio') return el.checked;
          return (el.value && el.value.trim()!=='');
        });
        if(active) btn.classList.add('active-filter'); else btn.classList.remove('active-filter');
      });
    }

    // --- gather config for backend ---
    function gatherConfig(mode){
      const cfg={
        mode:mode||'preview',
        course:document.getElementById('course').value,
        topicsRaw:document.getElementById('topics').value,
        excludeTopicsRaw:document.getElementById('exclude-topics').value,
        diffList:checkedValues('#difficulties'),
        typeList:checkedValues('#types'),
        corrList:checkedValues('#correct-status'),
        myAnsList:splitList(document.getElementById('my-answer').value),
        sourceRaw:document.getElementById('source').value,
        excludeSourceRaw:document.getElementById('exclude-source').value,
        searchText:document.getElementById('search-text').value,
        practicedWithinDays:document.getElementById('practiced-within').value,
        notPracticedDays:document.getElementById('not-practiced').value,
        isPracticeEmptyStr:document.getElementById('is-practice-empty').checked?'true':'false',
        examTitle:document.getElementById('exam-title').value,
        examSubtitle:document.getElementById('exam-subtitle').value,
        maxQ:document.getElementById('max-questions').value,
        shuffleChoices:document.getElementById('shuffle').checked,
        showTopicTag:document.getElementById('show-topic-tag').checked?'true':'false',
        includeSummary:document.getElementById('include-summary').checked?'true':'false',
        typeQuotas:{}, diffQuotas:{}
      };
      document.querySelectorAll('#type-quotas input').forEach(inp=>{
        if(inp.value){ cfg.typeQuotas[keyifyFromId(inp.id)] = inp.value; }
      });
      document.querySelectorAll('#diff-quotas input').forEach(inp=>{
        if(inp.value){ cfg.diffQuotas[keyifyFromId(inp.id)] = inp.value; }
      });
      return cfg;
    }
    function keyify(s){ return String(s).toLowerCase().replace(/[^a-z0-9]/g,''); }
    function keyifyFromId(id){ return String(id).replace('-quota',''); }
    function checkedValues(sel){ return Array.from(document.querySelectorAll(sel+' input[type=checkbox]:checked')).map(cb=>cb.value); }
    function splitList(s){ return !s?[]:s.split(/[;|,]/).map(x=>x.trim()).filter(Boolean); }

    // --- buttons wiring + handlers ---
    function wireButtons(){
      document.getElementById('btnPreview').addEventListener('click', previewCount);
      document.getElementById('btnReset').addEventListener('click', resetFilters);
      document.getElementById('btnStudent').addEventListener('click', ()=>generateExam('studentOnly'));
      document.getElementById('btnAnswer').addEventListener('click', ()=>generateExam('answerOnly'));
      document.getElementById('btnBoth').addEventListener('click', ()=>generateExam('bothOneDoc'));
    }

    function setStatus(msg, warn){
      const el=document.getElementById('status');
      el.textContent = msg || '';
      if(warn) el.classList.add('warn'); else el.classList.remove('warn');
    }

    function previewCount(){
      setStatus('Counting matches…');
      const cfg=gatherConfig('preview');
      google.script.run.withSuccessHandler(res=>{
        setStatus(res);
      }).withFailureHandler(err=>{
        setStatus('Error: '+err.message, true);
      }).previewCountFromUI(cfg);
    }

    function resetFilters(){
      // Core
      const courseSel=document.getElementById('course'); if(courseSel.options.length>0) courseSel.selectedIndex=0;
      document.getElementById('topics').value='';
      document.getElementById('exclude-topics').value='';
      // Content
      ['difficulties','types','correct-status'].forEach(id=>{
        document.querySelectorAll('#'+id+' input[type=checkbox]').forEach(cb=>cb.checked=false);
      });
      // Advanced
      document.getElementById('my-answer').value='';
      document.getElementById('source').value='';
      document.getElementById('exclude-source').value='';
      document.getElementById('search-text').value='';
      // Dates
      document.getElementById('practiced-within').value='';
      document.getElementById('not-practiced').value='';
      document.getElementById('is-practice-empty').checked=false;
      // Output
      document.getElementById('exam-title').value='';
      document.getElementById('exam-subtitle').value='';
      document.getElementById('max-questions').value='';
      document.getElementById('shuffle').checked=true;
      document.getElementById('show-topic-tag').checked=true;
      document.getElementById('include-summary').checked=false;
      // Quotas
      document.querySelectorAll('#type-quotas input,#diff-quotas input').forEach(i=>i.value='');
      updateActiveHighlights();
      setStatus('Filters were reset. Click Preview Count to see matches.');
    }

    function generateExam(mode){
      setStatus('Generating… please wait.');
      const cfg=gatherConfig(mode);
      google.script.run
        .withSuccessHandler(msg=>setStatus(msg))
        .withFailureHandler(err=>setStatus('Error: '+err.message, true))
        .buildExamFromUI(cfg);
    }
  </script>
